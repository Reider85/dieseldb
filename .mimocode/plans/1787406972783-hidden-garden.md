# Plan: Batch Index Insertion on Table Load (Prompt 55) — Optimized

## Problem
When a table is deserialized, secondary B-tree indexes are rebuilt one record at a time — O(N log N) per index. The clustered index already uses O(N) `bulkLoad()`, but `BTreeIndex` has no equivalent.

## Goal
Add `bulkLoad()` to `BTreeIndex` and use it in all bulk-rebuild paths. Change secondary BTree index construction from O(N log N) to O(N).

## Step 1: Add `bulkLoad()` to BTreeIndex.java

Insert after the `rangeSearchHigh` method (after line 315). The key difference from `BTreeClusteredIndex.bulkLoad()` is **duplicate key merging** — BTreeIndex stores `List<List<Integer>>` (multiple row indices per key), not `List<Integer>`.

```java
    /**
     * Bulk-loads the index from pre-sorted data. All key–rowIndex pairs are
     * supplied in ascending key order. Duplicate keys are merged into a single
     * entry with a combined list of row indices. After this call the previous
     * tree is discarded and replaced.
     *
     * @param sortedKeys   sorted keys (ascending)
     * @param sortedRowIdx corresponding row indices (same order as sortedKeys)
     * @throws IllegalArgumentException if lists are empty or have mismatched sizes
     */
    void bulkLoad(List<Object> sortedKeys, List<Integer> sortedRowIdx) {
        if (sortedKeys.size() != sortedRowIdx.size()) {
            throw new IllegalArgumentException("sortedKeys and sortedRowIdx must have the same size");
        }
        int n = sortedKeys.size();
        if (n == 0) {
            this.root = new Node(true);
            return;
        }

        // Merge duplicate keys into single entries with combined row index lists.
        List<Object> mergedKeys = new ArrayList<>();
        List<List<Integer>> mergedIndices = new ArrayList<>();
        Object prevKey = sortedKeys.get(0);
        List<Integer> currentIndices = new ArrayList<>();
        currentIndices.add(sortedRowIdx.get(0));
        for (int i = 1; i < n; i++) {
            Object key = sortedKeys.get(i);
            if (compareKeys(key, prevKey) == 0) {
                currentIndices.add(sortedRowIdx.get(i));
            } else {
                mergedKeys.add(prevKey);
                mergedIndices.add(currentIndices);
                prevKey = key;
                currentIndices = new ArrayList<>();
                currentIndices.add(sortedRowIdx.get(i));
            }
        }
        mergedKeys.add(prevKey);
        mergedIndices.add(currentIndices);

        // Build all leaf nodes from merged data (left to right).
        int leafCapacity = 2 * t - 1;
        List<Node> leaves = new ArrayList<>();
        Node currentLeaf = new Node(true);
        for (int i = 0; i < mergedKeys.size(); i++) {
            currentLeaf.keys.add(mergedKeys.get(i));
            currentLeaf.rowIndices.add(mergedIndices.get(i));
            if (currentLeaf.keys.size() == leafCapacity || i == mergedKeys.size() - 1) {
                leaves.add(currentLeaf);
                if (i < mergedKeys.size() - 1) {
                    currentLeaf = new Node(true);
                }
            }
        }

        // Build internal levels bottom-up.
        List<Node> currentLevel = leaves;
        while (currentLevel.size() > 1) {
            List<Node> nextLevel = new ArrayList<>();
            int i = 0;
            while (i < currentLevel.size()) {
                Node parent = new Node(false);
                parent.children.add(currentLevel.get(i));
                i++;
                while (parent.keys.size() < leafCapacity && i < currentLevel.size()) {
                    parent.keys.add(extractFirstKey(currentLevel.get(i)));
                    parent.children.add(currentLevel.get(i));
                    i++;
                }
                nextLevel.add(parent);
            }
            currentLevel = nextLevel;
        }

        this.root = currentLevel.get(0);
    }

    /**
     * Extracts the leftmost (smallest) key from a subtree.
     */
    private Object extractFirstKey(Node node) {
        if (node.isLeaf) {
            return node.keys.get(0);
        }
        return extractFirstKey(node.children.get(0));
    }
```

```java
package diesel;

import org.junit.jupiter.api.Test;
import java.util.*;
import static org.junit.jupiter.api.Assertions.*;

class BTreeIndexBulkLoadTest {

    @Test
    void emptyInput() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        idx.bulkLoad(List.of(), List.of());
        assertTrue(idx.search(1).isEmpty());
    }

    @Test
    void singleKey() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        idx.bulkLoad(List.of(5), List.of(0));
        assertEquals(List.of(0), idx.search(5));
        assertTrue(idx.search(1).isEmpty());
    }

    @Test
    void uniqueKeys() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        List<Object> keys = new ArrayList<>();
        List<Integer> rows = new ArrayList<>();
        for (int i = 0; i < 1000; i++) {
            keys.add(i);
            rows.add(i);
        }
        idx.bulkLoad(keys, rows);
        for (int i = 0; i < 1000; i++) {
            assertEquals(List.of(i), idx.search(i));
        }
        assertTrue(idx.search(-1).isEmpty());
    }

    @Test
    void duplicateKeys() {
        BTreeIndex idx = new BTreeIndex(String.class);
        List<Object> keys = List.of("a", "a", "a", "b", "b", "c");
        List<Integer> rows = List.of(0, 1, 2, 3, 4, 5);
        idx.bulkLoad(keys, rows);
        assertEquals(List.of(0, 1, 2), idx.search("a"));
        assertEquals(List.of(3, 4), idx.search("b"));
        assertEquals(List.of(5), idx.search("c"));
    }

    @Test
    void bulkLoadThenInsert() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        idx.bulkLoad(List.of(1, 3, 5), List.of(0, 1, 2));
        idx.insert(2, 3);
        assertEquals(List.of(3), idx.search(2));
        assertEquals(List.of(0), idx.search(1));
    }

    @Test
    void bulkLoadThenRemove() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        idx.bulkLoad(List.of(1, 2, 3), List.of(0, 1, 2));
        idx.remove(2, 1);
        assertTrue(idx.search(2).isEmpty());
    }

    @Test
    void mismatchedSizesThrows() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        assertThrows(IllegalArgumentException.class,
                () -> idx.bulkLoad(List.of(1, 2), List.of(0)));
    }

    @Test
    void rangeSearchAfterBulkLoad() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        List<Object> keys = new ArrayList<>();
        List<Integer> rows = new ArrayList<>();
        for (int i = 0; i < 100; i++) {
            keys.add(i);
            rows.add(i);
        }
        idx.bulkLoad(keys, rows);
        List<Integer> result = idx.rangeSearch(10, 20);
        assertEquals(11, result.size());
        for (int i = 10; i <= 20; i++) {
            assertTrue(result.contains(i));
        }
    }

    @Test
    void largeDataset() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        int N = 10_000;
        List<Object> keys = new ArrayList<>(N);
        List<Integer> rows = new ArrayList<>(N);
        for (int i = 0; i < N; i++) {
            keys.add(i);
            rows.add(i);
        }
        idx.bulkLoad(keys, rows);
        for (int i = 0; i < N; i++) {
            assertEquals(List.of(i), idx.search(i));
        }
    }

    @Test
    void duplicateRangeSearch() {
        BTreeIndex idx = new BTreeIndex(Integer.class);
        // Keys: 1,1,2,2,3,3
        List<Object> keys = List.of(1, 1, 2, 2, 3, 3);
        List<Integer> rows = List.of(0, 1, 2, 3, 4, 5);
        idx.bulkLoad(keys, rows);
        List<Integer> result = idx.rangeSearch(1, 2);
        assertEquals(4, result.size());
        assertTrue(result.containsAll(List.of(0, 1, 2, 3)));
```
